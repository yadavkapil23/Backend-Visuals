server.js (Express)

const express = require('express');
const app = express();

// 1. Define the Route with Path Variables (:origin, :dest)
app.post('/flights/:origin/:dest', (req, res) => {

    // 2. Extract Path Variables (The "Route")
    const { origin, dest } = req.params;
    console.log(`Searching flights from ${origin} to ${dest}`);

    // 3. Extract Query Parameters (The "Filters")
    const { sort, direct, date } = req.query;
    if (sort) console.log("Applying sorting...");
    
    // 4. Extract Body (The "Details")
    const { seatClass } = req.body;
    console.log(`Looking for ${seatClass} seats`);

    // Database Logic would go here...
    
    res.json({
        status: "success",
        route: `${origin} -> ${dest}`,
        price: 450
    });
});

req.query

Use for: Sorting, filtering, pagination.

?sort=asc&page=2

Example: Cheap flights only? Page 2 of results?

req.body

Use for: Large data, sensitive info, creation.
